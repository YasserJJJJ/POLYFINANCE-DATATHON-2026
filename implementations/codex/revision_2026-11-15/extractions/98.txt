livrable « prototype démontrable ».

file. La limite de 250 Mo une fois décompressé s'applique dans tous les cas.

Un exemple : expliquer les risques d'un actif selon un persona
import base64
import json
import boto3
bedrock = boto3.client("bedrock-runtime", region_name="us-east-1")
# Modèle peu coûteux : l'URL est publique, chaque appel est facturé à votre compte
MODEL_ID = "global.amazon.nova-2-lite-v1:0"
PERSONAS = {
    "prudent": "Prudent (tolérance au risque faible)",
    "equilibre": "Équilibré (tolérance au risque modérée)",
    "audacieux": "Audacieux (tolérance au risque élevée)",
}
def lambda_handler(event, context):
    body = event.get("body") or "{}"
    if event.get("isBase64Encoded"):
        body = base64.b64decode(body).decode("utf-8")
    try:
        data = json.loads(body)
    except json.JSONDecodeError:
        return _reponse(400, {"erreur": "Le corps de la requête doit être du JSON."})
    actif = str(data.get("actif", "")).strip()[:20]
    persona = str(data.get("persona", "")).strip().lower().replace("é", "e")
    if not actif or persona not in PERSONAS:
        return _reponse(400, {
            "erreur": "Champs attendus : actif (ex. XIU.TO) et persona (prudent, equilibre ou audacieux)."
        })
    prompt = (
        f"Expliquez en 5 phrases au plus les principaux risques de l'actif {actif} "
        f"pour un investisseur fictif au profil {PERSONAS[persona]}. "
        "Distinguez les faits généraux des hypothèses. "
        "Ne donnez aucune recommandation d'achat ou de vente et ne citez aucun prix."
    )
    response = bedrock.converse(
        modelId=MODEL_ID,
        messages=[{"role": "user", "content": [{"text": prompt}]}],
        inferenceConfig={"maxTokens": 400},
    )
    # Concaténer les blocs de texte : certains modèles renvoient d'abord un bloc de raisonnement
    blocs = response["output"]["message"]["content"]
    explication = "".join(b["text"] for b in blocs if "text" in b)
    print(json.dumps({"actif": actif, "persona": persona, "usage": response["usage"]}))
    return _reponse(200, {
        "actif": actif,
        "persona": persona,
        "explication": explication,
        "avertissement": "Simulation éducative. Ceci n'est pas un conseil financier.",
    })
def _reponse(code, contenu):
    return {
        "statusCode": code,
        "headers": {"Content-Type": "application/json; charset=utf-8"},
        "body": json.dumps(contenu, ensure_ascii=False),
    }

Le chemin le plus court : une fonction avec une URL publique
ACCOUNT=$(aws sts get-caller-identity --query Account --output text)
# 1. Un rôle d'exécution. Les comptes Workshop Studio permettent de le créer.
cat > /tmp/lambda-trust.json <<'EOF'
{"Version":"2012-10-17","Statement":[{"Effect":"Allow",
 "Principal":{"Service":"lambda.amazonaws.com"},"Action":"sts:AssumeRole"}]}
EOF
aws iam create-role --role-name datathon-lambda \
  --assume-role-policy-document file:///tmp/lambda-trust.json
aws iam attach-role-policy --role-name datathon-lambda \
  --policy-arn arn:aws:iam::aws:policy/service-role/AWSLambdaBasicExecutionRole
# Permission d'appeler les modèles Bedrock, et rien d'autre dans Bedrock
cat > /tmp/lambda-bedrock.json <<'EOF'
{"Version":"2012-10-17","Statement":[{"Effect":"Allow",
 "Action":["bedrock:InvokeModel","bedrock:InvokeModelWithResponseStream"],
 "Resource":"*"}]}
EOF
aws iam put-role-policy --role-name datathon-lambda \
  --policy-name bedrock-invoke \
  --policy-document file:///tmp/lambda-bedrock.json
sleep 10   # IAM a besoin d'un moment pour propager le rôle
# 2. La fonction
zip function.zip lambda_function.py
aws lambda create-function \
  --function-name datathon-demo \
  --runtime python3.13 \
  --role "arn:aws:iam::${ACCOUNT}:role/datathon-lambda" \
  --handler lambda_function.lambda_handler \
  --zip-file fileb://function.zip \
  --timeout 60 \
  --memory-size 512 \
  --region us-east-1
# 3. Une URL HTTPS publique
aws lambda create-function-url-config \
  --function-name datathon-demo \
  --auth-type NONE \
  --region us-east-1
# Deux autorisations distinctes sont nécessaires, sinon l'URL renvoie 403 Forbidden
aws lambda add-permission \
  --function-name datathon-demo \
  --statement-id public-url \
  --action lambda:InvokeFunctionUrl \
  --principal '*' \
  --function-url-auth-type NONE \
  --region us-east-1
aws lambda add-permission \
  --function-name datathon-demo \
  --statement-id public-url-invoke \
  --action lambda:InvokeFunction \
  --principal '*' \
  --invoked-via-function-url \
  --region us-east-1

--auth-type NONE signiﬁe que toute personne qui possède le lien peut appeler la fonction, et chaque appel est facturé à votre compte. C'est ce qui la rend pratique pour une
démo. N'y mettez donc jamais d'identiﬁants, de données personnelles ou de boucle d'appels coûteuse : gardez un modèle peu coûteux, une limite maxTokens et une validation des
entrées, comme dans l'exemple. Si le jury n'a pas besoin de cliquer sur le lien, l'option privée est plus sûre : présentez depuis votre portable, ou utilisez --auth-type AWS_IAM (les
appels doivent alors être signés avec des identiﬁants AWS). Supprimez l'URL publique dès qu'elle n'est plus utile.
L'URL cesse de fonctionner le lundi 16 novembre à 13 h 59, quand les comptes sont fermés et les données détruites. Montrez la démo dans votre vidéo et gardez le code dans
GitHub : ce sont eux qui restent.
